אֶלָּא אֵימָא, קִבֵּל שָׂכָר כְּנֶגֶד כֻּלָּן. קשא למה נקיט רבי יהושע בן לוי לשון דאיכא למיטעי ביה? לימא שכר כנגד כולן!ונראה לי בס"ד דנקיט שכר כולן ללמדנו לפי דרכו דבר אחד והוא דידוע חשבון האבנים טובות הוא כך דאם אבן טובה משקל אחד ערכה במאה דינר אם תהיה גדולה עשרה משקלים אין אומדים ערכה אלף דנרים אלא חושבין כל משקל אחד ממנה באלף דנרים ויהיה מחירה סך הכל עשרה אלפים כי כיון שהיא גדולה במשקל יגדל ערכה על דרך זה, וכן על זה הדרך כל מה שתהיה גדולה במשקלה יגדל ערכה וכן הוא חשבון שכר המצות וכמו שאמר התנא (משנה אבות ב, טז) 'אִם לָמַדְתָּ תוֹרָה הַרְבֵּה נוֹתְנִים לָךְ שָׂכָר הַרְבֵּה' וכאשר הבאתי הקדמה זו בסה"ק אֶבֶן שְׁלֹמֹה על פסוק (משלי ח, יט) 'טוֹב פִּרְיִי מֵחָרוּץ וּמִפָּז וּתְבוּאָתִי מִכֶּסֶף נִבְחָר'.וכן הענין כאן עשרה שקראו בתורה אם יהיה שכר כל אחד ואחד בפני עצמו אלף אז זה הגולל אף על פי שהוא אדם אחד לוקח שכר עשרה אלפים שהוא שיעור שכר כל העשרה לזה אמר לוקח שכר כולן כלומר שיעור שכר על כולן ואינו לוקח משלהם אלא מגזא דרחמנא נטיל וכל מה שירבו העולים יותר ירבה שכרו יותר כפי חשבון ערך אבנים טובות הנזכר לעיל.
שְׁנֵי תַּלְמִידֵי חֲכָמִים שֶׁיּוֹשְׁבִים בְּעִיר אַחַת, וְאֵין נוֹחִין זֶה לָזֶה בַּהֲלָכָה, עֲלֵיהֶם הַכָּתוּב אוֹמֵר: "וּמִשְׁפָּטִים לֹא יִחְיוּ בָּהֶם". פירוש אין נוחין מצד הלב שבלבבם רוצין כל אחד לסתור דברי חבירו עליהם נאמר (יחזקאל כ, כה) 'מִשְׁפָּטִים לֹא יִחְיוּ בָּהֶם' והטעם כי לימוד התורה נקרא טוֹב דכתיב (משלי ד, ב) 'כִּי לֶקַח טוֹב נָתַתִּי לָכֶם' ושני מיני לימוד שהם משנה וגמרא דהיינו גרסה וסברה וכל אחד נקרא טוֹב, וב' פעמים טוֹב [17] הוא דַּל [34], רמז שאם יהיו ב' לימודים אלו ביחיד הוא לימוד דַּל, אבל אם יהיו בשני חברים יהיה הלימוד אצלם מספר חָכָם [68] ומספר חַיִּים [68] כי ב' פעמים , דַּל עולה חַיִּים, ולכן אם שנים אלו אין נוחין זה לזה אין להם התחברות דַּל עם דַּל להשלים אצלם מספר חַיִּים, ולכן עליהם הוא אומר 'וּמִשְׁפָּטִים לֹא יִחְיוּ בָּהֶם' וזהו שנאמר (משלי ד, כב) 'כִּי חַיִּים הֵם לְמֹצְאֵיהֶם וּלְכָל בְּשָׂרוֹ מַרְפֵּא' ודרשו רז"ל (עירובין נג:) למוציאהם בפה, והיינו כי הלומד ביחידי אינו מוציא את הדברים בפיו אך הלומד עם חבירו מוכרח להוציא הדברים בפיו או הוא או חבירו ובכהאי גונא יהיה נשלם אצלם מספר חַיִּים.בְּשֵׁם ה' נַעֲשֶׂה וְנַצְלִיחַ לְשֵׁם יִחוּד קוּדְשָׁא בְּרִיךְ הוּא וּשְׁכִינְתֵּיהּ
